#### 任务调度队列

```go
// 任务类型
type TaskType string
const (
    TaskTypeScan      TaskType = "scan"
    TaskTypeBackup    TaskType = "backup"
    TaskTypeCleanup   TaskType = "cleanup"
)

// 任务优先级
type TaskPriority int
const (
    PriorityHigh   TaskPriority = 1   // 用户触发扫描
    PriorityNormal TaskPriority = 5   // 自动备份
    PriorityLow    TaskPriority = 10  // 定时清理
)

// 异步任务队列
type TaskScheduler struct {
    queue    chan *Task          // 任务队列
    workers  int                 // 工作协程数
    running  map[string]*Task    // 正在执行的任务
    mu       sync.RWMutex        // 保护running
}

// 任务执行流程
func (s *TaskScheduler) Start() {
    for i := 0; i < s.workers; i++ {
        go s.worker()  // 启动多个工作协程
    }
}

func (s *TaskScheduler) worker() {
    for task := range s.queue {
        s.executeTask(task)  // 阻塞式执行，确保资源控制
    }
}
```

## 五、关键实现细节

### 5.1 后端多线程安全

```go
// 设备管理器 - 线程安全
type DeviceManager struct {
    devices map[string]*Device
    mu      sync.RWMutex          // 读写锁保护devices
    
    monitors map[string]context.CancelFunc
    muMonitors sync.Mutex         // 保护monitors
}

func (m *DeviceManager) GetDevice(id string) (*Device, error) {
    m.mu.RLock()
    defer m.mu.RUnlock()
    
    device, ok := m.devices[id]
    if !ok {
        return nil, ErrDeviceNotFound
    }
    return device, nil
}

func (m *DeviceManager) UpdateDeviceStatus(id string, status DeviceStatus) error {
    m.mu.Lock()
    defer m.mu.Unlock()
    
    device, ok := m.devices[id]
    if !ok {
        return ErrDeviceNotFound
    }
    
    device.Status = status
    device.LastSeen = time.Now()
    
    // 异步保存到数据库，不阻塞锁
    go m.saveDeviceToDB(device)
    
    return nil
}
```

### 5.2 前端状态管理

```typescript
// Pinia Store示例 - 设备管理
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { deviceApi } from '@/api/device'

export const useDeviceStore = defineStore('devices', () => {
  // State
  const devices = ref<Device[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)
  
  // Getters
  const onlineDevices = computed(() => 
    devices.value.filter(d => d.status === 'online')
  )
  
  const deviceCount = computed(() => devices.value.length)
  
  // Actions
  async function fetchDevices() {
    loading.value = true
    error.value = null
    try {
      const response = await deviceApi.list()
      devices.value = response.data
    } catch (err) {
      error.value = err.message
      throw err
    } finally {
      loading.value = false
    }
  }
  
  async function addDevice(deviceData: CreateDeviceRequest) {
    const response = await deviceApi.create(deviceData)
    devices.value.push(response.data)
    return response.data
  }
  
  // WebSocket实时更新
  function updateDeviceStatus(deviceId: string, status: DeviceStatus) {
```

### 5.3 资源限制与熔断

```go
// 资源限制器
type ResourceLimiter struct {
    semScan   chan struct{}       // 扫描并发限制
    semFile   chan struct{}       // 文件处理并发限制
    semDB     chan struct{}       // 数据库并发限制
}

func NewResourceLimiter() *ResourceLimiter {
    return &ResourceLimiter{
        semScan: make(chan struct{}, 5),      // 最多5个并发扫描
        semFile: make(chan struct{}, 10),     // 最多10个并发文件处理
        semDB:   make(chan struct{}, 20),     // 最多20个并发DB操作
    }
}

func (l *ResourceLimiter) AcquireScan(ctx context.Context) error {
    select {
    case l.semScan <- struct{}{}:
        return nil
    case <-ctx.Done():
        return ctx.Err()
    }
}

func (l *ResourceLimiter) ReleaseScan() {
    <-l.semScan
}

// 使用示例
func (s *ScanService) ExecuteScan(ctx context.Context, task *ScanTask) error {
    // 获取资源许可
    if err := s.limitManager.AcquireScan(ctx); err != nil {
        return fmt.Errorf("acquire scan resource failed: %w", err)
    }
    defer s.limitManager.ReleaseScan()
    
    // 执行扫描...
}
```
